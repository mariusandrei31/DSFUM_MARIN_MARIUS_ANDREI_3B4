
float tensiune = 0;
int pin = 7;
unsigned long timpStart = 0;
int perioada = 2000;
int timpHIGH = 0;

void setup() {
  pinMode(pin, OUTPUT);
  Serial.begin(9600);
}

void loop() {
  if (Serial.available() > 0) {
    tensiune = Serial.parseFloat();

    if (tensiune < 0)
      tensiune = 0;
    if (tensiune > 5)
      tensiune = 5;

    timpHIGH = tensiune * perioada / 5;
  }

  unsigned long timp = micros() % perioada;

  if (timp < timpHIGH)
    digitalWrite(pin, HIGH);
  else
    digitalWrite(pin, LOW);
}
