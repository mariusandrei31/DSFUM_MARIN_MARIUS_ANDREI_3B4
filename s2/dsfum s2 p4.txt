// the setup routine runs once when you press reset:
int viteza=1000,Viz;
void setup() {
  // declare pin 9 to be an output:
  pinMode(9, OUTPUT);
  Serial.begin(9600);
  
}

// the loop routine runs over and over again forever:
void loop() {
  if (Serial.available() > 0) {
    viteza = Serial.parseInt(); 
    
  }
  int intensitate=0;
  int timpStart=millis();
  if(Viz!=viteza && viteza!=0)
    Viz=viteza;
  int copieV=Viz/100;
  int fader=255/copieV;
  while(copieV){
    if(timpStart+100==millis()){
      copieV--;
      analogWrite(9,intensitate);
      intensitate+=fader;
      timpStart+=100;
    }

  }
}
