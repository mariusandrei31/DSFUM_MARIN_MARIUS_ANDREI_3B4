void setup() {
    for(byte i=8; i<=13; i++)
        pinMode(i, OUTPUT);
}

void loop() {
  
    for(int i=1;i<=63;i++){
      int led[16];
      int k=7;
      int copieI=i;
      while(copieI){
        led[++k]=copieI%2;
        copieI/=2;
      }
      for(byte j=8;j<=k;j++){
        delay(500);
        digitalWrite(j,led[j]);
      }
    }
    
}
